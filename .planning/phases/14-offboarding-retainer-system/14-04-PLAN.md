Use lazy tab data loading pattern:
```typescript
const [checkIns, setCheckIns] = useState<RetainerCheckIn[]>([])
const [isLoading, setIsLoading] = useState(false)
const [dueInfo, setDueInfo] = useState<{ dueDate: string, isOverdue: boolean } | null>(null)

useEffect(() => {
  setIsLoading(true)
  Promise.all([
    getRetainerCheckIns(project.id),
    getNextCheckInDueDate(project.id)
  ])
    .then(([data, due]) => {
      setCheckIns(data)
      setDueInfo(due)
    })
    .catch(console.error)
    .finally(() => setIsLoading(false))
}, [project.id])
```
