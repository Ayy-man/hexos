import { revalidatePath } from 'next/cache'

  getInputHash,
  type BriefSourceType,
  type BriefExtraction,
} from '@/lib/api/brief-extractions'

**generateBriefAction(params):**
```typescript
export async function generateBriefAction(params: {
  sourceType: BriefSourceType
  sourceId: string
  sourceData: Record<string, unknown>
}): Promise<BriefExtraction>
```

Logic:
1. Generate input hash from sourceData: `const inputHash = await getInputHash(sourceData)`
2. Check cache: `const cached = await getCachedBrief(sourceType, sourceId)`
3. If cached exists and `cached.input_hash === inputHash`, return cached
4. If no cache or hash mismatch, call `/api/generate-brief` endpoint
5. Save result with `saveBriefExtraction({ sourceType, sourceId, briefContent, redactedBrief, inputHash, tokensUsed, generationTimeMs })`
6. Return the saved extraction

**regenerateBriefAction(params):**
```typescript
export async function regenerateBriefAction(params: {
  sourceType: BriefSourceType
  sourceId: string
  sourceData: Record<string, unknown>
}): Promise<BriefExtraction>
```

Logic:
1. Invalidate existing cache: `await invalidateBriefCache(sourceType, sourceId)`
2. Call generateBriefAction (which will now generate fresh)
3. Revalidate relevant paths

**getBriefForOpportunityAction(opportunityId):**
```typescript
export async function getBriefForOpportunityAction(
  opportunityId: string
): Promise<BriefExtraction | null>
```

Logic:
1. Get opportunity data from DB
2. Get linked project/inquiry data if available
3. Compile sourceData object
4. Call generateBriefAction with sourceType='opportunity'

This action assembles the data and delegates to generateBriefAction.
